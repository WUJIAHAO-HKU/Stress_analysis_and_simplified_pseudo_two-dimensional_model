function phie_fit(x::Union{Array{Float64},Float64}, Li::Array{Float64}, ki::Array{Float64}, phie0::Float64, typef::String="sin")
    Ln, Ls, Lp = Li
    kn, ks, kp = ki
    L = Ln + Ls + Lp
    phie = zeros(length(x))

    # 微调边界层参数
    δn = 0.0576 * Ln  # 仅微微调整边界层厚度
    δp = 0.0576 * Lp
    
    if typef == "sin"
        for i in eachindex(x)
            if x[i] <= Ln
                # 负极区域：微调四阶多项式系数
                t = x[i]/Ln
                phie[i] = kn * (0.5*t^2 - 0.3455*t^3 + 0.1055*t^4) - kn * Ln/2 - ks * Ls/2 + phie0
                
                # 边界层修正
                if x[i] >= Ln - δn
                    β = (x[i] - (Ln - δn))/δn
                    correction = 0.1585 * kn * δn * (β^2 * (3 - 2β) + 0.121*β*(1 - β)*(1 - 2β))
                    phie[i] += correction
                end
                
            elseif x[i] < Ln + Ls
                # 隔膜区域：微调过渡函数
                t = (x[i] - Ln)/Ls
                base = ks * (x[i] - Ln - Ls/2)
                transition = t^2 * (3 - 2t) + 0.121*t*(1 - t)*(1 - 2t)
                correction = 0.1585 * ks * Ls * (transition - 0.5)
                phie[i] = base + correction + phie0
                
            else
                # 正极区域：微调四阶多项式系数
                t = (x[i] - (Ln + Ls))/Lp
                phie[i] = -kp * (0.5*t^2 - 0.3455*t^3 + 0.1055*t^4) + kp * Lp/2 + ks * Ls/2 + phie0
                
                # 边界层修正
                if x[i] <= Ln + Ls + δp
                    β = (x[i] - (Ln + Ls))/δp
                    correction = 0.1585 * kp * δp * (β^2 * (3 - 2β) + 0.121*β*(1 - β)*(1 - 2β))
                    phie[i] += correction
                end
            end
            
        end
    elseif typef == "quad"
        pi = acos(-1.0)
        for i in eachindex(x)
            if x[i] <= Ln
                # 负极区域：微调谐波系数
                t = x[i]/Ln
                phie[i] = -2 * kn * Ln/pi * (
                    cos(pi*t/2) + 
                    0.1285*cos(3*pi*t/2) + 
                    0.0345*cos(5*pi*t/2)
                ) - ks * Ls/2 + phie0
                
                # 边界层修正
                if x[i] >= Ln - δn
                    β = (x[i] - (Ln - δn))/δn
                    correction = 0.1585 * kn * δn * (sin(pi*β) + 0.121*sin(2*pi*β)*(1 - β))
                    phie[i] += correction
                end
                
            elseif x[i] < Ln + Ls
                # 隔膜区域：微调过渡函数
                t = (x[i] - Ln)/Ls
                base = ks * (x[i] - Ln - Ls/2)
                transition = 0.5 * (1 - cos(pi*t)) + 0.0625 * sin(2*pi*t)*(1 - t)*t
                correction = 0.1585 * ks * Ls * (transition - 0.5)
                phie[i] = base + correction + phie0
                
            else
                # 正极区域：微调谐波系数
                t = (x[i] - (Ln + Ls))/Lp
                phie[i] = 2 * kp * Lp/pi * (
                    sin(pi*t/2) + 
                    0.1285*sin(3*pi*t/2) + 
                    0.0345*sin(5*pi*t/2)
                ) + ks * Ls/2 + phie0
                
                # 边界层修正
                if x[i] <= Ln + Ls + δp
                    β = (x[i] - (Ln + Ls))/δp
                    correction = 0.1585 * kn * δn * (sin(pi*β) + 0.121*sin(2*pi*β)*(1 - β))
                    phie[i] += correction
                end
            end
        end
    end

    # 微调边界处理
    for i in eachindex(x)
        if abs(x[i] - Ln) < 1e-10 || abs(x[i] - (Ln + Ls)) < 1e-10
            if i > 2 && i < length(x) - 1
                w1, w2, w3, w4 = 0.1165, 0.3835, 0.3835, 0.1165
                phie[i] = w1*phie[i-2] + w2*phie[i-1] + w3*phie[i+1] + w4*phie[i+2]
            end
        end
    end

    return phie
end

function phie_fit(x::Union{Array{Float64},Float64}, Li::Array{Float64}, ki::Array{Float64}, phie0::Float64, typef::String="sin")
    Ln, Ls, Lp = Li
    kn, ks, kp = ki
    L = Ln + Ls + Lp
    phie = zeros(length(x))

    # 进一步优化边界层参数
    δn = 0.0575 * Ln  # 微调边界层厚度
    δp = 0.0575 * Lp
    
    if typef == "sin"
        for i in eachindex(x)
            if x[i] <= Ln
                # 负极区域：进一步优化的多项式拟合
                t = x[i]/Ln
                # 精细调整多项式系数
                phie[i] = kn * (0.5*t^2 - 0.345*t^3 + 0.105*t^4) - kn * Ln/2 - ks * Ls/2 + phie0
                
                # 优化边界层修正
                if x[i] >= Ln - δn
                    β = (x[i] - (Ln - δn))/δn
                    # 改进的过渡函数
                    correction = 0.158 * kn * δn * (β^2 * (3 - 2β) + 0.12*β*(1 - β)*(1 - 2β))
                    phie[i] += correction
                end
                
            elseif x[i] < Ln + Ls
                # 隔膜区域：优化的非线性插值
                t = (x[i] - Ln)/Ls
                # 改进的过渡函数
                base = ks * (x[i] - Ln - Ls/2)
                transition = t^2 * (3 - 2t) + 0.12*t*(1 - t)*(1 - 2t)  # 进一步提升平滑度
                correction = 0.158 * ks * Ls * (transition - 0.5)
                phie[i] = base + correction + phie0
                
            else
                # 正极区域：进一步优化的多项式拟合
                t = (x[i] - (Ln + Ls))/Lp
                # 精细调整多项式系数
                phie[i] = -kp * (0.5*t^2 - 0.345*t^3 + 0.105*t^4) + kp * Lp/2 + ks * Ls/2 + phie0
                
                # 优化边界层修正
                if x[i] <= Ln + Ls + δp
                    β = (x[i] - (Ln + Ls))/δp
                    # 改进的过渡函数
                    correction = 0.158 * kp * δp * (β^2 * (3 - 2β) + 0.12*β*(1 - β)*(1 - 2β))
                    phie[i] += correction
                end
            end
            
        end
    elseif typef == "quad"
        pi = acos(-1.0)
        for i in eachindex(x)
            if x[i] <= Ln
                # 负极区域：优化的三角函数
                t = x[i]/Ln
                # 精细调整谐波系数
                phie[i] = -2 * kn * Ln/pi * (
                    cos(pi*t/2) + 
                    0.128*cos(3*pi*t/2) + 
                    0.034*cos(5*pi*t/2)
                ) - ks * Ls/2 + phie0
                
                # 优化边界层修正
                if x[i] >= Ln - δn
                    β = (x[i] - (Ln - δn))/δn
                    correction = 0.158 * kn * δn * (sin(pi*β) + 0.12*sin(2*pi*β)*(1 - β))
                    phie[i] += correction
                end
                
            elseif x[i] < Ln + Ls
                # 隔膜区域：优化的非线性插值
                t = (x[i] - Ln)/Ls
                # 改进的过渡函数
                base = ks * (x[i] - Ln - Ls/2)
                transition = 0.5 * (1 - cos(pi*t)) + 0.06 * sin(2*pi*t)*(1 - t)*t
                correction = 0.158 * ks * Ls * (transition - 0.5)
                phie[i] = base + correction + phie0
                
            else
                # 正极区域：优化的三角函数
                t = (x[i] - (Ln + Ls))/Lp
                # 精细调整谐波系数
                phie[i] = 2 * kp * Lp/pi * (
                    sin(pi*t/2) + 
                    0.128*sin(3*pi*t/2) + 
                    0.034*sin(5*pi*t/2)
                ) + ks * Ls/2 + phie0
                
                # 优化边界层修正
                if x[i] <= Ln + Ls + δp
                    β = (x[i] - (Ln + Ls))/δp
                    correction = 0.158 * kp * δp * (sin(pi*β) + 0.12*sin(2*pi*β)*(1 - β))
                    phie[i] += correction
                end
            end
        end
    end

    # 优化的边界处理
    for i in eachindex(x)
        if abs(x[i] - Ln) < 1e-10 || abs(x[i] - (Ln + Ls)) < 1e-10
            if i > 2 && i < length(x) - 1
                # 精细调整权重
                w1, w2, w3, w4 = 0.115, 0.385, 0.385, 0.115
                phie[i] = w1*phie[i-2] + w2*phie[i-1] + w3*phie[i+1] + w4*phie[i+2]
            end
        end
    end

    return phie
end

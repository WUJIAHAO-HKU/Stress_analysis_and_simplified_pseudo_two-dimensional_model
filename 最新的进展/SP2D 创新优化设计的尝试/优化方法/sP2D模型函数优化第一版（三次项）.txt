//主要改进包括：

//在“sin”模式中：

//添加三次项来提供更好的曲率控制

//在电极区域使用改进的多项式拟合

//在隔膜区域添加二次项实现平滑过渡

//在“quad”模式中：

//添加高阶谐波项以提高拟合精度

//改进三角函数的系数

//在隔膜区域添加平滑过渡项

//边界处理：

//添加边界点的特殊处理

//使用相邻点平均值确保连续性


function phie_fit(x::Union{Array{Float64},Float64}, Li::Array{Float64}, ki::Array{Float64}, phie0::Float64, typef::String="sin")
    Ln, Ls, Lp = Li
    kn, ks, kp = ki
    L = Ln + Ls + Lp
    phie = zeros(length(x))

    if typef == "sin"
        for i in eachindex(x)
            if x[i] <= Ln
                # 负极区域：改进的二次函数
                t = x[i]/Ln
                # 添加三次项来提供更好的曲率控制
                phie[i] = kn * (0.5 * t^2 - t^3/3) - kn * Ln/2 - ks * Ls/2 + phie0
            elseif x[i] < Ln + Ls
                # 隔膜区域：改进的线性插值
                t = (x[i] - Ln)/Ls
                # 添加平滑过渡
                phie[i] = ks * (x[i] - Ln - Ls/2) + 
                         0.1 * ks * Ls * (t^2 - t) + phie0
            else
                # 正极区域：改进的二次函数
                t = (x[i] - (Ln + Ls))/Lp
                # 添加三次项来提供更好的曲率控制
                phie[i] = -kp * (0.5 * t^2 - t^3/3) + kp * Lp/2 + ks * Ls/2 + phie0
            end
        end
    elseif typef == "quad"
        pi = acos(-1.0)
        for i in eachindex(x)
            if x[i] <= Ln
                # 负极区域：改进的余弦函数
                t = x[i]/Ln
                # 添加高阶谐波以提高精度
                phie[i] = -2 * kn * Ln/pi * (cos(pi*t/2) + 0.1*cos(3*pi*t/2)) - 
                         ks * Ls/2 + phie0
            elseif x[i] < Ln + Ls
                # 隔膜区域：改进的线性插值
                t = (x[i] - Ln)/Ls
                # 添加平滑过渡
                phie[i] = ks * (x[i] - Ln - Ls/2) + 
                         0.1 * ks * Ls * (t^2 - t) + phie0
            else
                # 正极区域：改进的正弦函数
                t = (x[i] - (Ln + Ls))/Lp
                # 添加高阶谐波以提高精度
                phie[i] = 2 * kp * Lp/pi * (sin(pi*t/2) + 0.1*sin(3*pi*t/2)) + 
                         ks * Ls/2 + phie0
            end
        end
    end

    # 边界处理以确保连续性
    for i in eachindex(x)
        if abs(x[i] - Ln) < 1e-10 || abs(x[i] - (Ln + Ls)) < 1e-10
            # 在界面处使用相邻点的平均值
            if i > 1 && i < length(x)
                phie[i] = (phie[i-1] + phie[i+1])/2
            end
        end
    end

    return phie
end

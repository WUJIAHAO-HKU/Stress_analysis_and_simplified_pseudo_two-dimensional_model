//主要改进包括：

//添加边界层处理：

//引入边界层参数δn和δp

//在边界层区域使用特殊的过渡函数

//添加局部修正项

//改进拟合函数：

//在“sin”模式中使用四阶多项式

//在“quad”模式中添加五阶谐波

//优化系数选择

//改进隔膜区域处理：

//使用更平滑的S型过渡函数

//添加非线性修正项

//优化过渡区域的连续性

//改进边界处理：

//使用四点加权平均

//优化权重分配

//扩大平滑区域

//这些改进应该能够：

//提供更精确的电位分布

//实现更平滑的界面过渡

//减少数值振荡

//保持物理合理性

function phie_fit(x::Union{Array{Float64},Float64}, Li::Array{Float64}, ki::Array{Float64}, phie0::Float64, typef::String="sin")
    Ln, Ls, Lp = Li
    kn, ks, kp = ki
    L = Ln + Ls + Lp
    phie = zeros(length(x))

    # 添加边界层参数来优化过渡区域
    δn = 0.05 * Ln  # 负极边界层厚度
    δp = 0.05 * Lp  # 正极边界层厚度
    
    if typef == "sin"
        for i in eachindex(x)
            if x[i] <= Ln
                # 负极区域：改进的多项式拟合
                t = x[i]/Ln
                # 使用四阶多项式提供更精确的控制
                phie[i] = kn * (0.5*t^2 - t^3/3 + t^4/12) - kn * Ln/2 - ks * Ls/2 + phie0
                
                # 在靠近隔膜的边界层添加修正
                if x[i] >= Ln - δn
                    β = (x[i] - (Ln - δn))/δn
                    correction = 0.15 * kn * δn * (β^2 * (3 - 2β))
                    phie[i] += correction
                end
                
            elseif x[i] < Ln + Ls
                # 隔膜区域：改进的非线性插值
                t = (x[i] - Ln)/Ls
                # 使用改进的过渡函数
                base = ks * (x[i] - Ln - Ls/2)
                transition = t^2 * (3 - 2t)  # 更平滑的S型函数
                correction = 0.15 * ks * Ls * (transition - 0.5)
                phie[i] = base + correction + phie0
                
            else
                # 正极区域：改进的多项式拟合
                t = (x[i] - (Ln + Ls))/Lp
                # 使用四阶多项式提供更精确的控制
                phie[i] = -kp * (0.5*t^2 - t^3/3 + t^4/12) + kp * Lp/2 + ks * Ls/2 + phie0
                
                # 在靠近隔膜的边界层添加修正
                if x[i] <= Ln + Ls + δp
                    β = (x[i] - (Ln + Ls))/δp
                    correction = 0.15 * kp * δp * (β^2 * (3 - 2β))
                    phie[i] += correction
                end
            end
            
        end
    elseif typef == "quad"
        pi = acos(-1.0)
        for i in eachindex(x)
            if x[i] <= Ln
                # 负极区域：改进的三角函数
                t = x[i]/Ln
                # 使用更复杂的三角函数组合
                phie[i] = -2 * kn * Ln/pi * (
                    cos(pi*t/2) + 
                    0.12*cos(3*pi*t/2) + 
                    0.03*cos(5*pi*t/2)
                ) - ks * Ls/2 + phie0
                
                # 边界层修正
                if x[i] >= Ln - δn
                    β = (x[i] - (Ln - δn))/δn
                    correction = 0.15 * kn * δn * sin(pi*β)
                    phie[i] += correction
                end
                
            elseif x[i] < Ln + Ls
                # 隔膜区域：改进的非线性插值
                t = (x[i] - Ln)/Ls
                # 使用改进的过渡函数
                base = ks * (x[i] - Ln - Ls/2)
                transition = 0.5 * (1 - cos(pi*t))  # 余弦过渡函数
                correction = 0.15 * ks * Ls * (transition - 0.5)
                phie[i] = base + correction + phie0
                
            else
                # 正极区域：改进的三角函数
                t = (x[i] - (Ln + Ls))/Lp
                # 使用更复杂的三角函数组合
                phie[i] = 2 * kp * Lp/pi * (
                    sin(pi*t/2) + 
                    0.12*sin(3*pi*t/2) + 
                    0.03*sin(5*pi*t/2)
                ) + ks * Ls/2 + phie0
                
                # 边界层修正
                if x[i] <= Ln + Ls + δp
                    β = (x[i] - (Ln + Ls))/δp
                    correction = 0.15 * kp * δp * sin(pi*β)
                    phie[i] += correction
                end
            end
        end
    end

    # 改进的边界处理
    for i in eachindex(x)
        if abs(x[i] - Ln) < 1e-10 || abs(x[i] - (Ln + Ls)) < 1e-10
            # 使用加权平均实现更平滑的过渡
            if i > 2 && i < length(x) - 1
                w1, w2, w3, w4 = 0.1, 0.4, 0.4, 0.1  # 权重
                phie[i] = w1*phie[i-2] + w2*phie[i-1] + w3*phie[i+1] + w4*phie[i+2]
            end
        end
    end

    return phie
end

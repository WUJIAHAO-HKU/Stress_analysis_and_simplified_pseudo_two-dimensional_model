function phie_fit(x::Union{Array{Float64},Float64}, Li::Array{Float64}, ki::Array{Float64}, phie0::Float64, typef::String="sin")
    Ln, Ls, Lp = Li
    kn, ks, kp = ki
    L = Ln + Ls + Lp
    phie = zeros(length(x))

    # 微调边界层参数
    δn = 0.0572 * Ln  # 微调边界层厚度
    δp = 0.0572 * Lp
    
    if typef == "sin"
        for i in eachindex(x)
            if x[i] <= Ln
                # 负极区域：五阶多项式拟合
                t = x[i]/Ln
                # 引入五阶项并优化系数
                phie[i] = kn * (0.5*t^2 - 0.342*t^3 + 0.108*t^4 - 0.018*t^5) - 
                         kn * Ln/2 - ks * Ls/2 + phie0
                
                # 优化边界层修正
                if x[i] >= Ln - δn
                    β = (x[i] - (Ln - δn))/δn
                    # 改进的过渡函数
                    correction = 0.157 * kn * δn * (β^2 * (3 - 2β) + 0.118*β*(1 - β)*(1 - 2β))
                    phie[i] += correction
                end
                
            elseif x[i] < Ln + Ls
                # 隔膜区域：优化的非线性插值
                t = (x[i] - Ln)/Ls
                # 改进的过渡函数
                base = ks * (x[i] - Ln - Ls/2)
                # 优化的五阶过渡函数
                transition = t^2 * (3 - 2t) + 0.118*t*(1 - t)*(1 - 2t) - 0.015*t*(1 - t)*(1 - 2t)^2
                correction = 0.157 * ks * Ls * (transition - 0.5)
                phie[i] = base + correction + phie0
                
            else
                # 正极区域：五阶多项式拟合
                t = (x[i] - (Ln + Ls))/Lp
                # 引入五阶项并优化系数
                phie[i] = -kp * (0.5*t^2 - 0.342*t^3 + 0.108*t^4 - 0.018*t^5) + 
                         kp * Lp/2 + ks * Ls/2 + phie0
                
                # 优化边界层修正
                if x[i] <= Ln + Ls + δp
                    β = (x[i] - (Ln + Ls))/δp
                    # 改进的过渡函数
                    correction = 0.157 * kp * δp * (β^2 * (3 - 2β) + 0.118*β*(1 - β)*(1 - 2β))
                    phie[i] += correction
                end
            end
            
        end
    elseif typef == "quad"
        pi = acos(-1.0)
        for i in eachindex(x)
            if x[i] <= Ln
                # 负极区域：优化的三角函数
                t = x[i]/Ln
                # 优化谐波系数并添加七次谐波
                phie[i] = -2 * kn * Ln/pi * (
                    cos(pi*t/2) + 
                    0.127*cos(3*pi*t/2) + 
                    0.0335*cos(5*pi*t/2) +
                    0.006*cos(7*pi*t/2)
                ) - ks * Ls/2 + phie0
                
                # 优化边界层修正
                if x[i] >= Ln - δn
                    β = (x[i] - (Ln - δn))/δn
                    correction = 0.157 * kn * δn * (sin(pi*β) + 0.118*sin(2*pi*β)*(1 - β))
                    phie[i] += correction
                end
                
            elseif x[i] < Ln + Ls
                # 隔膜区域：优化的非线性插值
                t = (x[i] - Ln)/Ls
                # 改进的过渡函数
                base = ks * (x[i] - Ln - Ls/2)
                transition = 0.5 * (1 - cos(pi*t)) + 0.062 * sin(2*pi*t)*(1 - t)*t - 
                           0.008 * sin(3*pi*t)*(1 - t)*t
                correction = 0.157 * ks * Ls * (transition - 0.5)
                phie[i] = base + correction + phie0
                
            else
                # 正极区域：优化的三角函数
                t = (x[i] - (Ln + Ls))/Lp
                # 优化谐波系数并添加七次谐波
                phie[i] = 2 * kp * Lp/pi * (
                    sin(pi*t/2) + 
                    0.127*sin(3*pi*t/2) + 
                    0.0335*sin(5*pi*t/2) +
                    0.006*sin(7*pi*t/2)
                ) + ks * Ls/2 + phie0
                
                # 优化边界层修正
                if x[i] <= Ln + Ls + δp
                    β = (x[i] - (Ln + Ls))/δp
                    correction = 0.157 * kp * δp * (sin(pi*β) + 0.118*sin(2*pi*β)*(1 - β))
                    phie[i] += correction
                end
            end
        end
    end

    # 优化的边界处理
    for i in eachindex(x)
        if abs(x[i] - Ln) < 1e-10 || abs(x[i] - (Ln + Ls)) < 1e-10
            if i > 2 && i < length(x) - 1
                # 微调权重
                w1, w2, w3, w4 = 0.116, 0.384, 0.384, 0.116
                phie[i] = w1*phie[i-2] + w2*phie[i-1] + w3*phie[i+1] + w4*phie[i+2]
            end
        end
    end

    return phie
end
